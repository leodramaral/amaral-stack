{{- /* Alerts (> [!NOTE] …), captioned quotes and plain quotes. Colors come from tokens: accent for note/important, ok for tip, warn for warning/caution. */ -}} {{- $alert := .AlertType -}} {{- $caption := .Attributes.caption -}} {{- $tone := "" -}} {{- $mark := "" -}} {{- if eq .Type "alert" -}} {{- if or (eq $alert "warning") (eq $alert "caution") -}}{{ $tone = "warn" }}{{ $mark = "!" }} {{- else if eq $alert "tip" -}}{{ $tone = "ok" }}{{ $mark = "↑" }} {{- else -}}{{ $tone = "accent" }}{{ $mark = "i" }} {{- end -}} {{- end -}} {{- if $tone -}} {{- $box := dict "warn" "border-warn/30 bg-warn/10" "ok" "border-ok/30 bg-ok/10" "accent" "border-accent/30 bg-accent-soft" -}} {{- $text := dict "warn" "text-warn" "ok" "text-ok" "accent" "text-accent" -}} {{- else -}} {{- end -}}
